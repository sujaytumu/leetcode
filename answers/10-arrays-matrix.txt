function solve(matrix) {
  const res = [];
  if (matrix.length === 0) return res;
  let top = 0, bottom = matrix.length - 1, left = 0, right = matrix[0].length - 1;
  while (top <= bottom && left <= right) {
    for (let j = left; j <= right; j++) res.push(matrix[top][j]);
    for (let i = top + 1; i <= bottom; i++) res.push(matrix[i][right]);
    if (top < bottom && left < right) {
      for (let j = right - 1; j >= left; j--) res.push(matrix[bottom][j]);
      for (let i = bottom - 1; i > top; i--) res.push(matrix[i][left]);
    }
    top++;
    bottom--;
    left++;
    right--;
  }
  return res;
}

    int n = nums.size();
    int best = nums[0] + nums[1] + nums[2];
    for (int i = 0; i < n - 2; i++) {
        int l = i + 1, r = n - 1;
        while (l < r) {
            int s = nums[i] + nums[l] + nums[r];
            if (abs(s - target) < abs(best - target)) best = s;
            if (s < target) l++;
            else if (s > target) r--;
            else return s;
        }
    }
    return best;
}
