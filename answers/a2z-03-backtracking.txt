void subsetsGo(const vector<int> &nums, int i, vector<int> &cur, vector<vector<int>> &out) {
    if (i == (int)nums.size()) {
        out.push_back(cur);
        return;
    }
    subsetsGo(nums, i + 1, cur, out);              // skip nums[i]
    cur.push_back(nums[i]);
    subsetsGo(nums, i + 1, cur, out);              // take nums[i]
    cur.pop_back();
}

vector<vector<int>> solve(vector<int> nums) {
    vector<vector<int>> out;
    vector<int> cur;
    subsetsGo(nums, 0, cur, out);
    return out;
}

void subsetsUniqueGo(const vector<int> &nums, int start, vector<int> &cur, vector<vector<int>> &out) {
    out.push_back(cur);
    for (int i = start; i < (int)nums.size(); i++) {
        if (i > start && nums[i] == nums[i - 1]) continue;   // same value at the same depth: skip
        cur.push_back(nums[i]);
        subsetsUniqueGo(nums, i + 1, cur, out);
        cur.pop_back();
    }
}

vector<vector<int>> solve(vector<int> nums) {
    sort(nums.begin(), nums.end());
    vector<vector<int>> out;
    vector<int> cur;
    subsetsUniqueGo(nums, 0, cur, out);
    return out;
}

void comboTwoGo(const vector<int> &c, int start, int left, vector<int> &cur, vector<vector<int>> &out) {
    if (left == 0) {
        out.push_back(cur);
        return;
    }
    for (int i = start; i < (int)c.size() && c[i] <= left; i++) {
        if (i > start && c[i] == c[i - 1]) continue;
        cur.push_back(c[i]);
        comboTwoGo(c, i + 1, left - c[i], cur, out);
        cur.pop_back();
    }
}

vector<vector<int>> solve(vector<int> candidates, int target) {
    sort(candidates.begin(), candidates.end());
    vector<vector<int>> out;
    vector<int> cur;
    comboTwoGo(candidates, 0, target, cur, out);
    return out;
}

void comboThreeGo(int next, int k, int left, vector<int> &cur, vector<vector<int>> &out) {
    if (k == 0) {
        if (left == 0) out.push_back(cur);
        return;
    }
    for (int d = next; d <= 9 && d <= left; d++) {
        cur.push_back(d);
        comboThreeGo(d + 1, k - 1, left - d, cur, out);
        cur.pop_back();
    }
}

vector<vector<int>> solve(int k, int n) {
    vector<vector<int>> out;
    vector<int> cur;
    comboThreeGo(1, k, n, cur, out);
    return out;
}

void permuteSwap(vector<int> &a, int i, vector<vector<int>> &out) {
    if (i == (int)a.size()) {
        out.push_back(a);
        return;
    }
    for (int j = i; j < (int)a.size(); j++) {
        swap(a[i], a[j]);
        permuteSwap(a, i + 1, out);
        swap(a[i], a[j]);
    }
}

vector<vector<int>> solve(vector<int> nums) {
    vector<vector<int>> out;
    permuteSwap(nums, 0, out);
    return out;
}
=== letter-combinations-of-a-phone-number

void phoneGo(const string &digits, int i, string &cur, const vector<string> &letters, vector<string> &out) {
    if (i == (int)digits.size()) {
        out.push_back(cur);
        return;
    }
    for (char c : letters[digits[i] - '0']) {
        cur.push_back(c);
        phoneGo(digits, i + 1, cur, letters, out);
        cur.pop_back();
    }
}

vector<string> solve(string digits) {
    vector<string> out;
    if (digits.empty()) return out;
    vector<string> letters = {"", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};
    string cur;
    phoneGo(digits, 0, cur, letters, out);
    return out;
}

vector<int> solve(vector<int> nums) {
    vector<int> sums = {0};
    for (int x : nums) {
        int size = sums.size();
        for (int i = 0; i < size; i++) sums.push_back(sums[i] + x);   // every old sum, with and without x
    }
    sort(sums.begin(), sums.end());
    return sums;
}

string solve(int n, int k) {
    vector<int> digits;
    int fact = 1;
    for (int i = 1; i < n; i++) {
        fact *= i;
        digits.push_back(i);
    }
    digits.push_back(n);
    k--;                                          // zero-based rank
    string out;
    while (true) {
        int idx = k / fact;
        out += to_string(digits[idx]);
        digits.erase(digits.begin() + idx);
        if (digits.empty()) break;
        k %= fact;
        fact /= digits.size();
    }
    return out;
}
=== rat-in-a-maze
--- cpp
void mazeGo(const vector<vector<int>> &maze, int r, int c, string &path, vector<vector<bool>> &seen, vector<string> &out) {
    int n = maze.size();
    if (r == n - 1 && c == n - 1) {
        out.push_back(path);
        return;
    }
    static const int dr[4] = {1, 0, 0, -1}, dc[4] = {0, -1, 1, 0};
    static const char mv[4] = {'D', 'L', 'R', 'U'};
    for (int d = 0; d < 4; d++) {
        int a = r + dr[d], b = c + dc[d];
        if (a < 0 || b < 0 || a >= n || b >= n || !maze[a][b] || seen[a][b]) continue;
        seen[a][b] = true;
        path.push_back(mv[d]);
        mazeGo(maze, a, b, path, seen, out);
        path.pop_back();
        seen[a][b] = false;
    }
}

vector<string> solve(vector<vector<int>> maze) {
    int n = maze.size();
    vector<string> out;
    if (!maze[0][0] || !maze[n - 1][n - 1]) return out;
    vector<vector<bool>> seen(n, vector<bool>(n, false));
    seen[0][0] = true;
    string path;
    mazeGo(maze, 0, 0, path, seen, out);
    return out;
}
=== m-coloring-problem

bool colorGo(const vector<vector<int>> &g, int v, int m, vector<int> &color) {
    int n = g.size();
    if (v == n) return true;
    for (int c = 1; c <= m; c++) {
        bool ok = true;
        for (int u = 0; u < v && ok; u++)
            if (g[v][u] && color[u] == c) ok = false;
        if (!ok) continue;
        color[v] = c;
        if (colorGo(g, v + 1, m, color)) return true;
        color[v] = 0;
    }
    return false;
}

bool solve(vector<vector<int>> graph, int m) {
    vector<int> color(graph.size(), 0);
    return colorGo(graph, 0, m, color);
}
=== binary-strings-without-adjacent-ones
--- cpp
void binaryGo(int n, string &cur, vector<string> &out) {
    if ((int)cur.size() == n) {
        out.push_back(cur);
        return;
    }
    cur.push_back('0');
    binaryGo(n, cur, out);
    cur.pop_back();
    if (cur.empty() || cur.back() != '1') {      // a 1 may only follow a 0 (or start the string)
        cur.push_back('1');
        binaryGo(n, cur, out);
        cur.pop_back();
    }
}

vector<string> solve(int n) {
    vector<string> out;
    string cur;
    binaryGo(n, cur, out);
    return out;
}
=== count-subsequences-with-sum-k
--- cpp
int solve(vector<int> nums, int k) {
    vector<long long> ways(k + 1, 0);              // ways[s] = subsequences seen so far that sum to s
    ways[0] = 1;
    for (int x : nums)
        for (int s = k; s >= x; s--) ways[s] += ways[s - x];
    return (int)ways[k];
}

void subseqGo(const vector<int> &nums, int i, int left, vector<int> &cur, vector<vector<int>> &out) {
    if (left == 0 && !cur.empty()) out.push_back(cur);
    if (left <= 0) return;
    for (int j = i; j < (int)nums.size(); j++) {
        if (nums[j] > left) continue;
        cur.push_back(nums[j]);
        subseqGo(nums, j + 1, left - nums[j], cur, out);
        cur.pop_back();
    }
}

vector<vector<int>> solve(vector<int> nums, int k) {
    vector<vector<int>> out;
    vector<int> cur;
    subseqGo(nums, 0, k, cur, out);
    return out;
}
