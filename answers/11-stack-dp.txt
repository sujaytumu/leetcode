def solve(arr):
    MOD = 10**9 + 7
    n = len(arr)
    left = [0] * n   # elements to the left (inclusive) where arr[i] is the strict minimum
    right = [0] * n  # elements to the right (inclusive) where arr[i] is the minimum (ties go left)
    st = []
    for i in range(n):
        while st and arr[st[-1]] > arr[i]:
            st.pop()
        left[i] = i - (st[-1] if st else -1)
        st.append(i)
    st = []
    for i in range(n - 1, -1, -1):
        while st and arr[st[-1]] >= arr[i]:
            st.pop()
        right[i] = (st[-1] if st else n) - i
        st.append(i)
    return sum(arr[i] * left[i] * right[i] for i in range(n)) % MOD

function solve(arr) {
  const MOD = 1000000007n;
  const n = arr.length;
  const left = new Array(n), right = new Array(n);
  let st = [];
  for (let i = 0; i < n; i++) {
    while (st.length && arr[st[st.length - 1]] > arr[i]) st.pop();
    left[i] = i - (st.length ? st[st.length - 1] : -1);
    st.push(i);
  }
  st = [];
  for (let i = n - 1; i >= 0; i--) {
    while (st.length && arr[st[st.length - 1]] >= arr[i]) st.pop();
    right[i] = (st.length ? st[st.length - 1] : n) - i;
    st.push(i);
  }
  let total = 0n; // BigInt: the sum can exceed 2^53
  for (let i = 0; i < n; i++) total += BigInt(arr[i]) * BigInt(left[i]) * BigInt(right[i]);
  return Number(total % MOD);
}
