=== three-sum
--- cpp
vector<vector<int>> solve(vector<int> nums) {
    sort(nums.begin(), nums.end());
    int n = nums.size();
    vector<vector<int>> out;
    for (int i = 0; i + 2 < n; i++) {
        if (i > 0 && nums[i] == nums[i - 1]) continue;       // skip duplicate first values
        int l = i + 1, r = n - 1;
        while (l < r) {
            int sum = nums[i] + nums[l] + nums[r];
            if (sum < 0) l++;
            else if (sum > 0) r--;
            else {
                out.push_back({nums[i], nums[l], nums[r]});
                l++; r--;
                while (l < r && nums[l] == nums[l - 1]) l++;   // skip duplicates on both sides
                while (l < r && nums[r] == nums[r + 1]) r--;
            }
        }
    }
    return out;
}
=== four-sum

vector<vector<int>> solve(vector<int> nums, int target) {
    sort(nums.begin(), nums.end());
    int n = nums.size();
    vector<vector<int>> out;
    for (int i = 0; i + 3 < n; i++) {
        if (i > 0 && nums[i] == nums[i - 1]) continue;
        for (int j = i + 1; j + 2 < n; j++) {
            if (j > i + 1 && nums[j] == nums[j - 1]) continue;
            int l = j + 1, r = n - 1;
            while (l < r) {
                long long sum = (long long)nums[i] + nums[j] + nums[l] + nums[r];
                if (sum < target) l++;
                else if (sum > target) r--;
                else {
                    out.push_back({nums[i], nums[j], nums[l], nums[r]});
                    l++; r--;
                    while (l < r && nums[l] == nums[l - 1]) l++;
                    while (l < r && nums[r] == nums[r + 1]) r--;
                }
            }
        }
    }
    return out;
}

bool solve(int n, vector<vector<int>> edges) {
    vector<int> parent(n);
    iota(parent.begin(), parent.end(), 0);
    function<int(int)> find = [&](int x) {
        while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }
        return x;
    };
    for (auto &e : edges) {
        int a = find(e[0]), b = find(e[1]);
        if (a == b) return true;          // both ends already connected: this edge closes a cycle
        parent[a] = b;
    }
    return false;
}
=== detect-cycle-in-a-directed-graph
--- cpp
bool solve(int n, vector<vector<int>> edges) {
    vector<vector<int>> adj(n);
    vector<int> indeg(n, 0);
    for (auto &e : edges) { adj[e[0]].push_back(e[1]); indeg[e[1]]++; }
    queue<int> q;
    for (int i = 0; i < n; i++) if (indeg[i] == 0) q.push(i);
    int done = 0;
    while (!q.empty()) {                  // Kahn's algorithm: nodes on a cycle never reach in-degree 0
        int u = q.front(); q.pop();
        done++;
        for (int v : adj[u]) if (--indeg[v] == 0) q.push(v);
    }
    return done < n;
}

vector<int> solve(int n, vector<vector<int>> edges) {
    vector<vector<int>> adj(n);
    vector<int> indeg(n, 0);
    for (auto &e : edges) { adj[e[0]].push_back(e[1]); indeg[e[1]]++; }
    priority_queue<int, vector<int>, greater<int>> heap;   // smallest available label first
    for (int i = 0; i < n; i++) if (indeg[i] == 0) heap.push(i);
    vector<int> order;
    while (!heap.empty()) {
        int u = heap.top(); heap.pop();
        order.push_back(u);
        for (int v : adj[u]) if (--indeg[v] == 0) heap.push(v);
    }
    return order;
}

=== shortest-path-in-undirected-graph
--- cpp
vector<int> solve(int n, vector<vector<int>> edges, int src) {
    vector<vector<int>> adj(n);
    for (auto &e : edges) { adj[e[0]].push_back(e[1]); adj[e[1]].push_back(e[0]); }
    vector<int> dist(n, -1);
    queue<int> q;
    dist[src] = 0;
    q.push(src);
    while (!q.empty()) {                  // BFS gives the fewest edges to every node
        int u = q.front(); q.pop();
        for (int v : adj[u]) if (dist[v] < 0) { dist[v] = dist[u] + 1; q.push(v); }
    }
    return dist;
}
=== dijkstra-algorithm
--- cpp
vector<int> solve(int n, vector<vector<int>> edges, int src) {
    vector<vector<pair<int, int>>> adj(n);
    for (auto &e : edges) { adj[e[0]].push_back({e[1], e[2]}); adj[e[1]].push_back({e[0], e[2]}); }
    const long long INF = LLONG_MAX;
    vector<long long> dist(n, INF);
    priority_queue<pair<long long, int>, vector<pair<long long, int>>, greater<>> heap;
    dist[src] = 0;
    heap.push({0, src});
    while (!heap.empty()) {
        auto [d, u] = heap.top(); heap.pop();
        if (d > dist[u]) continue;        // stale entry
        for (auto [v, w] : adj[u]) {
            if (d + w < dist[v]) { dist[v] = d + w; heap.push({dist[v], v}); }
        }
    }
    vector<int> out(n);
    for (int i = 0; i < n; i++) out[i] = dist[i] == INF ? -1 : (int)dist[i];
    return out;
}
=== bellman-ford-algorithm
--- cpp
vector<int> solve(int n, vector<vector<int>> edges, int src) {
    const long long INF = 100000000;
    vector<long long> dist(n, INF);
    dist[src] = 0;
    for (int round = 0; round < n - 1; round++) {          // n - 1 rounds are enough without a negative cycle
        bool changed = false;
        for (auto &e : edges) {
            if (dist[e[0]] != INF && dist[e[0]] + e[2] < dist[e[1]]) { dist[e[1]] = dist[e[0]] + e[2]; changed = true; }
        }
        if (!changed) break;
    }
    for (auto &e : edges) {                                // one more improvement means a reachable negative cycle
        if (dist[e[0]] != INF && dist[e[0]] + e[2] < dist[e[1]]) return {-1};
    }
    vector<int> out(n);
    for (int i = 0; i < n; i++) out[i] = (int)dist[i];
    return out;
}
=== floyd-warshall
--- cpp
vector<vector<int>> solve(vector<vector<int>> matrix) {
    int n = matrix.size();
    const long long INF = (long long)1e15;
    vector<vector<long long>> d(n, vector<long long>(n, INF));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) if (matrix[i][j] >= 0) d[i][j] = matrix[i][j];
    for (int i = 0; i < n; i++) d[i][i] = 0;
    for (int k = 0; k < n; k++)                            // allow node k as an intermediate stop
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (d[i][k] < INF && d[k][j] < INF) d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
    vector<vector<int>> out(n, vector<int>(n));
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) out[i][j] = d[i][j] >= INF ? -1 : (int)d[i][j];
    return out;
}
=== minimum-spanning-tree

int solve(int n, vector<vector<int>> edges) {
    sort(edges.begin(), edges.end(), [](const vector<int> &a, const vector<int> &b) { return a[2] < b[2]; });
    vector<int> parent(n);
    iota(parent.begin(), parent.end(), 0);
    function<int(int)> find = [&](int x) {
        while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }
        return x;
    };
    int total = 0;
    for (auto &e : edges) {               // Kruskal: take the cheapest edge that joins two different components
        int a = find(e[0]), b = find(e[1]);
        if (a != b) { parent[a] = b; total += e[2]; }
    }
    return total;
}

vector<int> solve(int n, vector<vector<int>> edges) {
    vector<vector<int>> rev(n);
    vector<int> outdeg(n, 0);
    for (auto &e : edges) { rev[e[1]].push_back(e[0]); outdeg[e[0]]++; }
    queue<int> q;
    for (int i = 0; i < n; i++) if (outdeg[i] == 0) q.push(i);   // terminal nodes are safe
    vector<int> safe;
    while (!q.empty()) {                  // a node becomes safe once all of its out-neighbours are safe
        int u = q.front(); q.pop();
        safe.push_back(u);
        for (int p : rev[u]) if (--outdeg[p] == 0) q.push(p);
    }
    sort(safe.begin(), safe.end());
    return safe;
}
=== number-of-operations-to-make-network-connected
--- cpp
int solve(int n, vector<vector<int>> connections) {
    if ((int)connections.size() < n - 1) return -1;          // not enough cables to join everything
    vector<int> parent(n);
    iota(parent.begin(), parent.end(), 0);
    function<int(int)> find = [&](int x) {
        while (parent[x] != x) { parent[x] = parent[parent[x]]; x = parent[x]; }
        return x;
    };
    int components = n;
    for (auto &c : connections) {
        int a = find(c[0]), b = find(c[1]);
        if (a != b) { parent[a] = b; components--; }
    }
    return components - 1;                // each spare cable can join one more component
}
=== strongly-connected-components-kosaraju
--- cpp
int solve(int n, vector<vector<int>> edges) {
    vector<vector<int>> adj(n), rev(n);
    for (auto &e : edges) { adj[e[0]].push_back(e[1]); rev[e[1]].push_back(e[0]); }
    vector<bool> seen(n, false);
    vector<int> order;
    function<void(int)> first = [&](int u) {              // pass 1: nodes by finishing time
        seen[u] = true;
        for (int v : adj[u]) if (!seen[v]) first(v);
        order.push_back(u);
    };
    for (int i = 0; i < n; i++) if (!seen[i]) first(i);
    fill(seen.begin(), seen.end(), false);
    function<void(int)> second = [&](int u) {             // pass 2: walk the reversed graph
        seen[u] = true;
        for (int v : rev[u]) if (!seen[v]) second(v);
    };
    int count = 0;
    for (int i = n - 1; i >= 0; i--) {
        if (!seen[order[i]]) { count++; second(order[i]); }
    }
    return count;
}
=== critical-connections-in-a-network
--- cpp
vector<vector<int>> solve(int n, vector<vector<int>> edges) {
    vector<vector<int>> adj(n);
    for (auto &e : edges) { adj[e[0]].push_back(e[1]); adj[e[1]].push_back(e[0]); }
    vector<int> tin(n, -1), low(n, 0);
    int timer = 0;
    vector<vector<int>> bridges;
    function<void(int, int)> dfs = [&](int u, int parent) {
        tin[u] = low[u] = timer++;
        for (int v : adj[u]) {
            if (v == parent) continue;
            if (tin[v] != -1) {
                low[u] = min(low[u], tin[v]);             // back edge
            } else {
                dfs(v, u);
                low[u] = min(low[u], low[v]);
                if (low[v] > tin[u]) bridges.push_back({min(u, v), max(u, v)});   // v's subtree cannot reach above u
            }
        }
    };
    for (int i = 0; i < n; i++) if (tin[i] == -1) dfs(i, -1);
    sort(bridges.begin(), bridges.end());
    return bridges;
}

vector<int> solve(int n, vector<vector<int>> edges) {
    vector<vector<int>> adj(n);
    for (auto &e : edges) { adj[e[0]].push_back(e[1]); adj[e[1]].push_back(e[0]); }
    vector<int> tin(n, -1), low(n, 0);
    vector<bool> isPoint(n, false);
    int timer = 0;
    function<void(int, int)> dfs = [&](int u, int parent) {
        tin[u] = low[u] = timer++;
        int children = 0;
        for (int v : adj[u]) {
            if (v == parent) continue;
            if (tin[v] != -1) {
                low[u] = min(low[u], tin[v]);
            } else {
                dfs(v, u);
                low[u] = min(low[u], low[v]);
                if (low[v] >= tin[u] && parent != -1) isPoint[u] = true;   // v's subtree hangs on u alone
                children++;
            }
        }
        if (parent == -1 && children > 1) isPoint[u] = true;                // a root with 2+ DFS children
    };
    for (int i = 0; i < n; i++) if (tin[i] == -1) dfs(i, -1);
    vector<int> out;
    for (int i = 0; i < n; i++) if (isPoint[i]) out.push_back(i);
    if (out.empty()) out.push_back(-1);
    return out;
}

=== construct-binary-search-tree-from-preorder-traversal
--- cpp
TreeNode* solve(vector<int> preorder) {
    int i = 0;
    // build(upper): consume values below `upper` for this subtree
    function<TreeNode*(long long)> build = [&](long long upper) -> TreeNode* {
        if (i == (int)preorder.size() || preorder[i] > upper) return nullptr;
        TreeNode* node = new TreeNode(preorder[i++]);
        node->left = build(node->val);    // the left subtree only holds smaller values
        node->right = build(upper);       // the right subtree keeps the parent's bound
        return node;
    };
    return build(LLONG_MAX);
}

bool solve(TreeNode* root, int k) {
    vector<int> vals;
    function<void(TreeNode*)> inorder = [&](TreeNode* n) {
        if (!n) return;
        inorder(n->left); vals.push_back(n->val); inorder(n->right);
    };
    inorder(root);                         // sorted values
    int l = 0, r = (int)vals.size() - 1;
    while (l < r) {
        int sum = vals[l] + vals[r];
        if (sum == k) return true;
        if (sum < k) l++; else r--;
    }
    return false;
}
=== recover-binary-search-tree
--- cpp
TreeNode* solve(TreeNode* root) {
    TreeNode *first = nullptr, *second = nullptr, *prev = nullptr;
    function<void(TreeNode*)> inorder = [&](TreeNode* n) {
        if (!n) return;
        inorder(n->left);
        if (prev && prev->val > n->val) {  // an inversion in the inorder sequence
            if (!first) first = prev;
            second = n;
        }
        prev = n;
        inorder(n->right);
    };
    inorder(root);
    if (first && second) swap(first->val, second->val);
    return root;
}
=== largest-bst-subtree
--- cpp
int solve(TreeNode* root) {
    int best = 0;
    // returns {isBST, size, smallest, largest} of the subtree
    function<array<long long, 4>(TreeNode*)> go = [&](TreeNode* n) -> array<long long, 4> {
        if (!n) return {1, 0, LLONG_MAX, LLONG_MIN};
        auto l = go(n->left), r = go(n->right);
        if (l[0] && r[0] && l[3] < n->val && n->val < r[2]) {
            int size = (int)(l[1] + r[1] + 1);
            best = max(best, size);
            return {1, size, min(l[2], (long long)n->val), max(r[3], (long long)n->val)};
        }
        return {0, 0, 0, 0};
    };
    go(root);
    return best;
}
=== amount-of-time-for-binary-tree-to-be-infected
--- cpp
int solve(TreeNode* root, int start) {
    unordered_map<TreeNode*, TreeNode*> parent;
    TreeNode* source = nullptr;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {                   // record every node's parent so the fire can climb up
        TreeNode* node = q.front(); q.pop();
        if (node->val == start) source = node;
        if (node->left) { parent[node->left] = node; q.push(node->left); }
        if (node->right) { parent[node->right] = node; q.push(node->right); }
    }
    unordered_set<TreeNode*> burnt{source};
    q.push(source);
    int minutes = -1;
    while (!q.empty()) {
        int size = q.size();
        minutes++;
        while (size--) {
            TreeNode* node = q.front(); q.pop();
            for (TreeNode* next : {node->left, node->right, parent.count(node) ? parent[node] : nullptr}) {
                if (next && !burnt.count(next)) { burnt.insert(next); q.push(next); }
            }
        }
    }
    return minutes;
}
=== all-nodes-distance-k-in-binary-tree
--- cpp
vector<int> solve(TreeNode* root, int target, int k) {
    unordered_map<TreeNode*, TreeNode*> parent;
    TreeNode* source = nullptr;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        TreeNode* node = q.front(); q.pop();
        if (node->val == target) source = node;
        if (node->left) { parent[node->left] = node; q.push(node->left); }
        if (node->right) { parent[node->right] = node; q.push(node->right); }
    }
    unordered_set<TreeNode*> seen{source};
    q.push(source);
    for (int step = 0; step < k && !q.empty(); step++) {   // move one edge further per round
        int size = q.size();
        while (size--) {
            TreeNode* node = q.front(); q.pop();
            for (TreeNode* next : {node->left, node->right, parent.count(node) ? parent[node] : nullptr}) {
                if (next && !seen.count(next)) { seen.insert(next); q.push(next); }
            }
        }
    }
    vector<int> out;
    while (!q.empty()) { out.push_back(q.front()->val); q.pop(); }
    sort(out.begin(), out.end());
    return out;
}
