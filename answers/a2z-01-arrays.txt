=== second-largest-element
--- cpp
int solve(vector<int> nums) {
    long long first = LLONG_MIN, second = LLONG_MIN;
    for (int x : nums) {
        if (x > first) {
            second = first;
            first = x;
        } else if (x < first && x > second) {
            second = x;
        }
    }
    return second == LLONG_MIN ? -1 : (int)second;
}

bool solve(vector<int> nums) {
    int n = nums.size(), drops = 0;
    for (int i = 0; i < n; i++)
        if (nums[i] > nums[(i + 1) % n]) drops++;
    return drops <= 1;
}

=== union-of-two-sorted-arrays
--- cpp
vector<int> solve(vector<int> a, vector<int> b) {
    vector<int> out;
    size_t i = 0, j = 0;
    auto add = [&](int x) {
        if (out.empty() || out.back() != x) out.push_back(x);
    };
    while (i < a.size() && j < b.size()) add(a[i] <= b[j] ? a[i++] : b[j++]);
    while (i < a.size()) add(a[i++]);
    while (j < b.size()) add(b[j++]);
    return out;
}

vector<int> solve(vector<int> nums) {

    size_t pos = 0, neg = 1;
    for (int x : nums) {
        if (x > 0) {

vector<int> solve(vector<int> nums) {
    vector<int> out;
    int best = INT_MIN;
    for (int i = (int)nums.size() - 1; i >= 0; i--) {
        if (nums[i] >= best) {
            out.push_back(nums[i]);
            best = nums[i];
        }
    }
    reverse(out.begin(), out.end());
    return out;
}
=== longest-subarray-with-sum-k
--- cpp
int solve(vector<int> nums, int k) {
    unordered_map<long long, int> first;
    first[0] = -1;
    long long total = 0;
    int best = 0;
    for (int i = 0; i < (int)nums.size(); i++) {
        total += nums[i];
        auto it = first.find(total - k);
        if (it != first.end()) best = max(best, i - it->second);
        first.emplace(total, i);
    }
    return best;
}
=== find-the-repeating-and-missing-number
--- cpp
vector<int> solve(vector<int> nums) {
    long long n = nums.size(), sum = 0, squares = 0;
    for (int x : nums) {
        sum += x;
        squares += 1LL * x * x;
    }
    long long diff = sum - n * (n + 1) / 2;                       // repeating - missing
    long long sumOf = (squares - n * (n + 1) * (2 * n + 1) / 6) / diff;  // repeating + missing
    long long repeating = (diff + sumOf) / 2;
    return {(int)repeating, (int)(repeating - diff)};
}
=== count-subarrays-with-given-xor
--- cpp
int solve(vector<int> nums, int k) {
    unordered_map<int, int> seen;
    seen[0] = 1;
    int xr = 0, total = 0;
    for (int x : nums) {
        xr ^= x;
        auto it = seen.find(xr ^ k);
        if (it != seen.end()) total += it->second;
        seen[xr]++;
    }
    return total;
}
=== count-inversions
--- cpp
long long mergeCount(vector<int> &a, vector<int> &tmp, int lo, int hi) {
    if (hi - lo < 2) return 0;
    int mid = (lo + hi) / 2;
    long long count = mergeCount(a, tmp, lo, mid) + mergeCount(a, tmp, mid, hi);
    int i = lo, j = mid, k = lo;
    while (i < mid && j < hi) {
        if (a[i] <= a[j]) {
            tmp[k++] = a[i++];
        } else {
            count += mid - i;
            tmp[k++] = a[j++];
        }
    }
    while (i < mid) tmp[k++] = a[i++];
    while (j < hi) tmp[k++] = a[j++];
    copy(tmp.begin() + lo, tmp.begin() + hi, a.begin() + lo);
    return count;
}

int solve(vector<int> nums) {
    vector<int> tmp(nums.size());
    return (int)mergeCount(nums, tmp, 0, nums.size());
}
=== reverse-pairs

long long pairsSort(vector<int> &a, vector<int> &tmp, int lo, int hi) {
    if (hi - lo < 2) return 0;
    int mid = (lo + hi) / 2;
    long long count = pairsSort(a, tmp, lo, mid) + pairsSort(a, tmp, mid, hi);
    int j = mid;
    for (int i = lo; i < mid; i++) {
        while (j < hi && (long long)a[i] > 2LL * a[j]) j++;
        count += j - mid;
    }
    merge(a.begin() + lo, a.begin() + mid, a.begin() + mid, a.begin() + hi, tmp.begin() + lo);
    copy(tmp.begin() + lo, tmp.begin() + hi, a.begin() + lo);
    return count;
}

int solve(vector<int> nums) {
    vector<int> tmp(nums.size());
    return (int)pairsSort(nums, tmp, 0, nums.size());
}
=== majority-element-ii
--- cpp
vector<int> solve(vector<int> nums) {
    int c1 = 0, c2 = 0, v1 = 0, v2 = 1;
    for (int x : nums) {
        if (x == v1) c1++;
        else if (x == v2) c2++;
        else if (c1 == 0) { v1 = x; c1 = 1; }
        else if (c2 == 0) { v2 = x; c2 = 1; }
        else { c1--; c2--; }
    }
    int n1 = 0, n2 = 0, n = nums.size();
    for (int x : nums) {
        if (x == v1) n1++;
        else if (x == v2) n2++;
    }
    vector<int> out;
    if (n1 > n / 3) out.push_back(v1);
    if (n2 > n / 3) out.push_back(v2);
    sort(out.begin(), out.end());
    return out;
}

int solve(vector<int> nums) {
    unordered_map<long long, int> first;
    first[0] = -1;
    long long total = 0;
    int best = 0;
    for (int i = 0; i < (int)nums.size(); i++) {
        total += nums[i];
        auto it = first.find(total);
        if (it != first.end()) best = max(best, i - it->second);

}
