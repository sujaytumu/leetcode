  const dist = Array.from({ length: m }, () => new Array(n).fill(-1));

    const [i, j] = q[h];

      const a = i + di, b = j + dj;
      if (a >= 0 && a < m && b >= 0 && b < n && dist[a][b] < 0) {
        dist[a][b] = dist[i][j] + 1;
        q.push([a, b]);
      }
    }
  }
  return dist;

                    color[v] = 1 - color[u]
                    stack.append(v)

import heapq


def solve(heights):
    m, n = len(heights), len(heights[0])
    best = [[float("inf")] * n for _ in range(m)]
    best[0][0] = 0
    pq = [(0, 0, 0)]
    while pq:
        e, i, j = heapq.heappop(pq)
        if e > best[i][j]:
            continue
        if i == m - 1 and j == n - 1:
            return e
        for di, dj in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            a, b = i + di, j + dj
            if 0 <= a < m and 0 <= b < n:
                ne = max(e, abs(heights[a][b] - heights[i][j]))
                if ne < best[a][b]:
                    best[a][b] = ne
                    heapq.heappush(pq, (ne, a, b))
    return 0

function solve(heights) {
  const m = heights.length, n = heights[0].length;
  const best = Array.from({ length: m }, () => new Array(n).fill(Infinity));
  best[0][0] = 0;
  // simple binary-heap-free approach: Bellman-Ford style relaxation with a queue (SPFA)
  const q = [[0, 0]];
  const dirs = [[1, 0], [-1, 0], [0, 1], [0, -1]];
  for (let h = 0; h < q.length; h++) {
    const [i, j] = q[h];
    for (const [di, dj] of dirs) {
      const a = i + di, b = j + dj;
      if (a < 0 || a >= m || b < 0 || b >= n) continue;
      const ne = Math.max(best[i][j], Math.abs(heights[a][b] - heights[i][j]));
      if (ne < best[a][b]) { best[a][b] = ne; q.push([a, b]); }
    }
  }
  return best[m - 1][n - 1];
}

def solve(grid):
    n = len(grid)
    seen = [[False] * n for _ in range(n)]
    pq = [(grid[0][0], 0, 0)]
    seen[0][0] = True
    while pq:
        t, i, j = heapq.heappop(pq)
        if i == n - 1 and j == n - 1:
            return t

            a, b = i + di, j + dj
            if 0 <= a < n and 0 <= b < n and not seen[a][b]:
                seen[a][b] = True
                heapq.heappush(pq, (max(t, grid[a][b]), a, b))
    return 0

function solve(grid) {
  const n = grid.length;
  // binary search on time, BFS feasibility check
  const ok = (T) => {
    if (grid[0][0] > T) return false;
    const seen = Array.from({ length: n }, () => new Array(n).fill(false));
    const st = [[0, 0]];
    seen[0][0] = true;
    while (st.length) {
      const [i, j] = st.pop();
      if (i === n - 1 && j === n - 1) return true;
      for (const [di, dj] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
        const a = i + di, b = j + dj;
        if (a >= 0 && a < n && b >= 0 && b < n && !seen[a][b] && grid[a][b] <= T) {
          seen[a][b] = true;
          st.push([a, b]);
        }
      }
    }
    return false;
  };
  let lo = 0, hi = 1;
  for (const r of grid) for (const v of r) hi = Math.max(hi, v);
  while (lo < hi) {
    const mid = (lo + hi) >> 1;
    if (ok(mid)) hi = mid; else lo = mid + 1;
  }
  return lo;
}

function solve(root) {
  if (!root) return [];
  const res = [];
  let level = [root];
  let ltr = true;
  while (level.length) {
    const vals = level.map((n) => n.val);
    res.push(ltr ? vals : vals.reverse());
    ltr = !ltr;
    const next = [];
    for (const n of level) {
      if (n.left) next.push(n.left);

    level = next;
  }
  return res;

    for c in t:
        need[c] = need.get(c, 0) + 1
    missing = len(t)
    best = 0
    left = 0
    for right, c in enumerate(s):
        if need.get(c, 0) > 0:
            missing -= 1
        need[c] = need.get(c, 0) - 1
        if missing == 0:
            while need[s[left]] < 0:

                left += 1
            length = right - left + 1

                best = length
            need[s[left]] += 1
            missing += 1
            left += 1
    return best

function solve(s, t) {
  const need = {};
  for (const c of t) need[c] = (need[c] || 0) + 1;
  let missing = t.length, best = 0, left = 0;
  for (let right = 0; right < s.length; right++) {
    const c = s[right];
    if ((need[c] || 0) > 0) missing--;
    need[c] = (need[c] || 0) - 1;
    if (missing === 0) {
      while (need[s[left]] < 0) { need[s[left]]++; left++; }
      const len = right - left + 1;
      if (best === 0 || len < best) best = len;
      need[s[left]]++;
      missing++;
      left++;
    }
  }
  return best;
}

static int solve(String s, String t) {
    int[] need = new int[128];
    for (char c : t.toCharArray()) need[c]++;
    int missing = t.length(), best = 0, left = 0;
    for (int right = 0; right < s.length(); right++) {
        char c = s.charAt(right);
        if (need[c] > 0) missing--;
        need[c]--;
        if (missing == 0) {
            while (need[s.charAt(left)] < 0) { need[s.charAt(left)]++; left++; }
            int len = right - left + 1;
            if (best == 0 || len < best) best = len;
            need[s.charAt(left)]++;
            missing++;
            left++;
        }
    }
    return best;
}
