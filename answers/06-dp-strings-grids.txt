def solve(nums):
    total = sum(nums)
    if total % 2:
        return False
    reachable = 1  # bit s is set when some subset sums to s
    for x in nums:
        reachable |= reachable << x
    return bool(reachable >> (total // 2) & 1)

        count, run = 1, 0
        for x in nums:

                run += x
