function solve(nums, target) {
  const seen = new Map();
  for (let i = 0; i < nums.length; i++) {
    const need = target - nums[i];
    if (seen.has(need)) return [seen.get(need), i];
    seen.set(nums[i], i);
  }
  return [];
}

def solve(s):
    pairs = {')': '(', ']': '[', '}': '{'}
    stack = []
    for c in s:
        if c in pairs:
            if not stack or stack.pop() != pairs[c]:
                return False

            stack.append(c)

--- js
function solve(s) {
  const pairs = { ")": "(", "]": "[", "}": "{" };
  const stack = [];
  for (const c of s) {
    if (c in pairs) {
      if (stack.length === 0 || stack.pop() !== pairs[c]) return false;
    } else {
      stack.push(c);
    }
  }
  return stack.length === 0;
}
--- cpp
bool solve(string s) {
    string stack;
    for (char c : s) {
        if (c == '(' || c == '[' || c == '{') {
            stack.push_back(c);
        } else {
            if (stack.empty()) return false;
            char open = stack.back();
            stack.pop_back();
            if ((c == ')' && open != '(') || (c == ']' && open != '[') || (c == '}' && open != '{')) return false;
        }
    }
    return stack.empty();
}
--- java
static boolean solve(String s) {
    Deque<Character> stack = new ArrayDeque<>();
    for (char c : s.toCharArray()) {
        if (c == '(' || c == '[' || c == '{') {
            stack.push(c);
        } else {
            if (stack.isEmpty()) return false;
            char open = stack.pop();
            if ((c == ')' && open != '(') || (c == ']' && open != '[') || (c == '}' && open != '{')) return false;
        }
    }
    return stack.isEmpty();
}

=== number-of-islands
--- py
def solve(grid):
    grid = [list(row) for row in grid]
    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    count = 0
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] != '1':
                continue
            count += 1
            grid[r][c] = '0'
            stack = [(r, c)]
            while stack:
                y, x = stack.pop()
                for ny, nx in ((y + 1, x), (y - 1, x), (y, x + 1), (y, x - 1)):
                    if 0 <= ny < rows and 0 <= nx < cols and grid[ny][nx] == '1':
                        grid[ny][nx] = '0'
                        stack.append((ny, nx))
    return count
--- js
function solve(grid) {
  const g = grid.map((row) => row.split(""));
  const rows = g.length, cols = rows ? g[0].length : 0;
  let count = 0;
  for (let r = 0; r < rows; r++) {
    for (let c = 0; c < cols; c++) {
      if (g[r][c] !== "1") continue;
      count++;
      g[r][c] = "0";
      const stack = [[r, c]];
      while (stack.length) {
        const [y, x] = stack.pop();
        for (const [ny, nx] of [[y + 1, x], [y - 1, x], [y, x + 1], [y, x - 1]]) {
          if (ny >= 0 && ny < rows && nx >= 0 && nx < cols && g[ny][nx] === "1") {
            g[ny][nx] = "0";
            stack.push([ny, nx]);
          }
        }
      }
    }
  }
  return count;
}
--- cpp
int solve(vector<string> grid) {
    int rows = grid.size(), cols = rows ? grid[0].size() : 0, count = 0;
    int dy[4] = {1, -1, 0, 0}, dx[4] = {0, 0, 1, -1};
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] != '1') continue;
            count++;
            grid[r][c] = '0';
            vector<pair<int, int>> stack = {{r, c}};
            while (!stack.empty()) {
                auto [y, x] = stack.back();
                stack.pop_back();
                for (int k = 0; k < 4; k++) {
                    int ny = y + dy[k], nx = x + dx[k];
                    if (ny >= 0 && ny < rows && nx >= 0 && nx < cols && grid[ny][nx] == '1') {
                        grid[ny][nx] = '0';
                        stack.push_back({ny, nx});
                    }
                }
            }
        }
    }
    return count;
}
--- java
static int solve(String[] grid) {
    int rows = grid.length, cols = rows > 0 ? grid[0].length() : 0, count = 0;
    char[][] g = new char[rows][];
    for (int i = 0; i < rows; i++) g[i] = grid[i].toCharArray();
    int[] dy = {1, -1, 0, 0}, dx = {0, 0, 1, -1};
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (g[r][c] != '1') continue;
            count++;
            g[r][c] = '0';
            Deque<int[]> stack = new ArrayDeque<>();
            stack.push(new int[]{r, c});
            while (!stack.isEmpty()) {
                int[] cur = stack.pop();
                for (int k = 0; k < 4; k++) {
                    int ny = cur[0] + dy[k], nx = cur[1] + dx[k];
                    if (ny >= 0 && ny < rows && nx >= 0 && nx < cols && g[ny][nx] == '1') {
                        g[ny][nx] = '0';
                        stack.push(new int[]{ny, nx});
                    }
                }
            }
        }
    }
    return count;
}

def solve(x):
    if x < 0 or (x % 10 == 0 and x != 0):

    rev = 0
    while x > rev:
        rev = rev * 10 + x % 10

function solve(x) {
  if (x < 0 || (x % 10 === 0 && x !== 0)) return false;
  let rev = 0;
  while (x > rev) {
    rev = rev * 10 + (x % 10);
    x = Math.floor(x / 10);
  }
  return x === rev || x === Math.floor(rev / 10);
}
